/**
 * Splits an integer amount of money (in minor units, e.g. kobo or cents)
 * among multiple recipients according to their fractional shares.
 *
 * Uses the largest-remainder (Hamilton) method so the allocations always
 * sum EXACTLY to `totalUnits` — no unit is lost or invented. Each recipient
 * first receives `floor(totalUnits * share / sumShares)` units; any leftover
 * units are then awarded one-by-one to the recipients with the largest
 * fractional remainders. Ties are broken deterministically by the
 * recipients' original order in `shares`.
 *
 * @param {number} totalUnits - Amount to allocate, as a non-negative integer number of minor units.
 * @param {number[]} shares - Relative weights for each recipient; a non-empty array of non-negative finite numbers whose sum is positive.
 * @returns {number[]} Integer allocations in the same order as `shares`, summing exactly to `totalUnits`.
 * @throws {RangeError} If `totalUnits` is not a non-negative integer, `shares` is empty, contains a negative/non-finite value, or its sum is not positive.
 *
 * @example
 * allocate(100, [1, 1, 1]);        // [34, 33, 33]
 * allocate(10, [1, 1, 1]);         // [4, 3, 3]
 * allocate(5, [3, 2]);             // [3, 2]
 */
function allocate(totalUnits, shares) {
  if (!Number.isInteger(totalUnits) || totalUnits < 0) {
    throw new RangeError('totalUnits must be a non-negative integer');
  }
  if (!Array.isArray(shares) || shares.length === 0) {
    throw new RangeError('shares must be a non-empty array');
  }

  let sumShares = 0;
  for (const s of shares) {
    if (typeof s !== 'number' || !Number.isFinite(s) || s < 0) {
      throw new RangeError('each share must be a non-negative finite number');
    }
    sumShares += s;
  }
  if (sumShares <= 0) {
    throw new RangeError('sum of shares must be positive');
  }

  const allocations = new Array(shares.length);
  const remainders = new Array(shares.length);
  let allocated = 0;

  for (let i = 0; i < shares.length; i++) {
    const exact = (totalUnits * shares[i]) / sumShares;
    const floor = Math.floor(exact);
    allocations[i] = floor;
    remainders[i] = exact - floor;
    allocated += floor;
  }

  const leftover = totalUnits - allocated;
  if (leftover > 0) {
    // Rank indices by descending remainder, then by ascending original index.
    const order = shares.map((_, i) => i);
    order.sort((a, b) => {
      if (remainders[b] !== remainders[a]) return remainders[b] - remainders[a];
      return a - b;
    });
    for (let k = 0; k < leftover; k++) {
      allocations[order[k]] += 1;
    }
  }

  return allocations;
}

module.exports = { allocate };
