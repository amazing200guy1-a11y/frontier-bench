/**
 * Extracts the first valid JSON object from arbitrary text.
 *
 * Searches for JSON objects embedded in prose, Markdown fences, or other
 * surrounding content. Handles nested objects, escaped quotes, and braces
 * inside JSON strings.
 *
 * @param {unknown} text - The text that may contain a JSON object.
 * @returns {Record<string, unknown> | null} The first valid JSON object,
 *   or null if no valid object is found.
 */
function extractJson(text) {
  if (typeof text !== 'string' || text.length === 0) {
    return null;
  }

  for (let start = 0; start < text.length; start++) {
    if (text[start] !== '{') {
      continue;
    }

    let depth = 0;
    let inString = false;
    let escaped = false;

    for (let end = start; end < text.length; end++) {
      const char = text[end];

      if (inString) {
        if (escaped) {
          escaped = false;
        } else if (char === '\\') {
          escaped = true;
        } else if (char === '"') {
          inString = false;
        }

        continue;
      }

      if (char === '"') {
        inString = true;
      } else if (char === '{') {
        depth++;
      } else if (char === '}') {
        depth--;

        if (depth === 0) {
          try {
            const value = JSON.parse(text.slice(start, end + 1));

            if (
              value !== null &&
              typeof value === 'object' &&
              !Array.isArray(value)
            ) {
              return value;
            }
          } catch {
            // This candidate is invalid; continue searching.
          }

          break;
        }
      }
    }
  }

  return null;
}

module.exports = { extractJson };
